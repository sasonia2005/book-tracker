def validate_date(date_string: str) -> Optional[str]:
    """
    Проверяет, существует ли дата.
    
    Args:
        date_string: Дата в формате YYYY-MM-DD или DD.MM.YYYY
        
    Returns:
        str: Дата в формате YYYY-MM-DD или None если дата некорректна
        
    Examples:
        >>> validate_date("2024-12-25")
        '2024-12-25'
        >>> validate_date("25.12.2024")
        '2024-12-25'
        >>> validate_date("32.13.2024")
        None
        >>> validate_date("12.45.2009")  # 45 дня не существует
        None
    """
    if not date_string:
        return None
    
    # Пробуем формат YYYY-MM-DD
    try:
        parsed = datetime.strptime(date_string, "%Y-%m-%d")
        # Дополнительная проверка на валидность (strptime уже проверяет)
        return parsed.strftime("%Y-%m-%d")
    except ValueError:
        pass
    
    # Пробуем формат DD.MM.YYYY
    try:
        parsed = datetime.strptime(date_string, "%d.%m.%Y")
        return parsed.strftime("%Y-%m-%d")
    except ValueError:
        pass
    
    logger.warning(f"Некорректная дата: {date_string}")
    return None


def get_valid_date(prompt: str = "Введите дату прочтения") -> str:
    """
    Запрашивает у пользователя дату и проверяет её корректность.
    Повторяет запрос до тех пор, пока не будет введена корректная дата.
    
    Args:
        prompt: Текст приглашения для ввода
        
    Returns:
        str: Корректная дата в формате YYYY-MM-DD
        
    Examples:
        >>> # Пользователь вводит "32.13.2024" -> ошибка, повторный запрос
        >>> # Пользователь вводит "25.12.2024" -> успех, возвращает "2024-12-25"
    """
    print(f"\n📅 {prompt}")
    print("   📌 Поддерживаемые форматы:")
    print("      • ГГГГ-ММ-ДД (например: 2024-12-25)")
    print("      • ДД.ММ.ГГГГ (например: 25.12.2024)")
    print("   💡 Если оставить пустым - будет использована сегодняшняя дата")
    print()
    
    while True:
        date_input = input("👉 Введите дату: ").strip()
        
        # Если пользователь ничего не ввел, используем сегодняшнюю дату
        if not date_input:
            today = datetime.now().strftime("%Y-%m-%d")
            print(f"   ✅ Использована сегодняшняя дата: {today}\n")
            return today
        
        # Проверяем корректность даты
        validated_date = validate_date(date_input)
        
        if validated_date:
            print(f"   ✅ Дата принята: {validated_date}\n")
            return validated_date
        else:
            print("\n   ❌ ОШИБКА: Введена некорректная дата!")
            print(f"   🔍 Вы ввели: '{date_input}'")
            print("   📋 Примеры корректных дат:")
            print("      • 25.12.2024 (25 декабря 2024 года)")
            print("      • 2024-12-25 (25 декабря 2024 года)")
            print("      • 29.02.2024 (29 февраля 2024 - високосный год)")
            print("   ❌ Некорректные примеры:")
            print("      • 32.13.2024 (32 дня, 13 месяцев)")
            print("      • 31.02.2024 (31 февраля не существует)")
            print("      • 12.45.2009 (45 дней не существует)")
            print("\n   🔄 Пожалуйста, попробуйте снова:\n")

class Book:
    """Модель книги."""
    
    def __init__(self, author: str, title: str, rating: int, date_read: str):
        """
        Инициализация книги.
        
        Args:
            author: Имя автора
            title: Название книги
            rating: Оценка от 1 до 5
            date_read: Дата прочтения в формате YYYY-MM-DD
        """
        self.author = author
        self.title = title
        self.rating = rating
        self.date_read = date_read
    
    def to_dict(self) -> Dict:
        """Конвертирует книгу в словарь для JSON."""
        return {
            "author": self.author,
            "title": self.title,
            "rating": self.rating,
            "date_read": self.date_read
        }
    
    @classmethod
    def from_dict(cls, data: Dict) -> 'Book':
        """Создает книгу из словаря."""
        return cls(
            author=data["author"],
            title=data["title"],
            rating=data["rating"],
            date_read=data["date_read"]
        )
    
    def __str__(self) -> str:
        """Строковое представление книги."""
        return f"{self.title} — {self.author}, оценка: {self.rating}, дата: {self.date_read}"


class BookRepository:
    """Репозиторий для работы с хранилищем книг."""
    
    def __init__(self, file_name: str = FILE_NAME):
        self.file_name = file_name
        logger.info(f"Инициализация репозитория с файлом {file_name}")
    
    def load_books(self) -> List[Book]:
        """
        Загружает книги из JSON файла.
        
        Returns:
            List[Book]: Список книг
        """
        try:
            with open(self.file_name, "r", encoding="utf-8") as f:
                data = json.load(f)
                logger.info(f"Загружено {len(data)} книг")
                return [Book.from_dict(book) for book in data]
        except FileNotFoundError:
            logger.warning(f"Файл {self.file_name} не найден, создан пустой список")
            return []
        except json.JSONDecodeError as e:
            logger.error(f"Ошибка парсинга JSON: {e}")
            return []
    
    def save_books(self, books: List[Book]) -> None:
        """
        Сохраняет книги в JSON файл.
        
        Args:
            books: Список книг для сохранения
        """
        try:
            data = [book.to_dict() for book in books]
            with open(self.file_name, "w", encoding="utf-8") as f:
                json.dump(data, f, ensure_ascii=False, indent=4)
            logger.info(f"Сохранено {len(books)} книг")
        except IOError as e:
            logger.error(f"Ошибка сохранения файла: {e}")
            raise

class BookService:
    """Сервис для операций с книгами."""
    
    def __init__(self, repository: BookRepository):
        self.repository = repository
    
    def add_book(self, author: str, title: str, rating: int, date_read: str) -> Optional[Book]:
        """
        Добавляет новую книгу.
        
        Args:
            author: Имя автора
            title: Название книги
            rating: Оценка от 1 до 5
            date_read: Дата прочтения (уже должна быть валидирована)
            
        Returns:
            Book: Добавленная книга или None при ошибке
        """
        # Валидация данных
        if not author or not title:
            logger.warning("Попытка добавить книгу без автора или названия")
            print("❌ Ошибка: автор и название не могут быть пустыми!")
            return None
        
        if rating not in VALID_RATINGS:
            logger.warning(f"Некорректная оценка: {rating}")
            print("❌ Ошибка: оценка должна быть от 1 до 5!")
            return None
        
        # Проверка на дубликат
        books = self.repository.load_books()
        if any(book.author == author and book.title == title for book in books):
            logger.info(f"Книга '{title}' уже существует")
            print(f"⚠️  Книга '{title}' автора {author} уже есть в списке!")
            return None
        
        new_book = Book(author, title, rating, date_read)
        books.append(new_book)
        self.repository.save_books(books)
        logger.info(f"Книга '{title}' добавлена успешно")
        return new_book
    
    def get_all_books(self) -> List[Book]:
        """Возвращает все книги."""
        return self.repository.load_books()
    
    def get_average_rating(self) -> float:
        """
        Возвращает среднюю оценку всех книг.
        
        Returns:
            float: Средняя оценка (0.0 если книг нет)
        """
        books = self.repository.load_books()
        if not books:
            return 0.0
        return sum(book.rating for book in books) / len(books)
    
    def get_author_stats(self) -> Dict[str, int]:
        """
        Возвращает статистику по авторам.
        
        Returns:
            Dict[str, int]: Словарь {автор: количество книг}
        """
        books = self.repository.load_books()
        stats = {}
        for book in books:
            stats[book.author] = stats.get(book.author, 0) + 1
        return stats
    
    def delete_book(self, index: int) -> bool:
        """
        Удаляет книгу по индексу.
        
        Args:
            index: Индекс книги в списке (начиная с 0)
            
        Returns:
            bool: True если удаление успешно, иначе False
        """
        books = self.repository.load_books()
        if 0 <= index < len(books):
            removed = books.pop(index)
            self.repository.save_books(books)
            logger.info(f"Книга '{removed.title}' удалена")
            print(f"✅ Книга '{removed.title}' успешно удалена!")
            return True
        logger.warning(f"Попытка удалить несуществующий индекс: {index}")
        print("❌ Неверный номер книги!")
        return False
    
    def search_books(self, query: str) -> List[Book]:
        """
        Поиск книг по автору или названию.
        
        Args:
            query: Поисковый запрос
            
        Returns:
            List[Book]: Список найденных книг
        """
        books = self.repository.load_books()
        query_lower = query.lower()
        results = [
            book for book in books
            if query_lower in book.title.lower() or query_lower in book.author.lower()
        ]
        logger.info(f"Поиск '{query}': найдено {len(results)} книг")
        return results

    def add_book_interactive(self):

        print("\n--- Добавление новой книги ---")
        
        author = input("Введите автора: ").strip()

            print("❌ Автор не может быть пустым!")
            return
        
        title = input("Введите название книги: ").strip()
        if not title:
            print("❌ Название не может быть пустым!")
            return
        
        # Ввод оценки с проверкой
        while True:
            try:
                rating = int(input("Введите оценку (1–5): "))
                if 1 <= rating <= 5:
                    break

            return
        
        print("\n" + "=" * 60)
        print(f"📚 СПИСОК ПРОЧИТАННЫХ КНИГ (всего: {len(books)})")

        """Показывает среднюю оценку."""
        books = self.service.get_all_books()
        
        if not books:
            print("\n📭 Список книг пуст, средняя оценка недоступна.")
            return
        
        avg = self.service.get_average_rating()
        print(f"\n⭐ Средняя оценка всех книг: {avg:.2f}")
        
        # Дополнительная статистика
        ratings_distribution = {}
        for book in books:
            ratings_distribution[book.rating] = ratings_distribution.get(book.rating, 0) + 1
        
        print("📊 Распределение оценок:")
        for rating in range(5, 0, -1):
            count = ratings_distribution.get(rating, 0)
            bar = "█" * count if count > 0 else "░"
            print(f"   {rating}⭐: {bar} ({count})")
    
    def show_author_stats_interactive(self):
        """Показывает статистику по авторам."""
        stats = self.service.get_author_stats()
        
        if not stats:
            print("\n📭 Список книг пуст, статистика по авторам недоступна.")
            return
        
        print("\n" + "=" * 50)

            print(f"   {author}: {count} {books_word}")

        print("-" * 50)

def main():
    """Главная функция приложения."""
    try:
        app = BookTrackerCLI()
        app.run()
    except KeyboardInterrupt:
        print("\n\n👋 Программа прервана пользователем. До свидания!")
        logger.info("Программа прервана пользователем")
        sys.exit(0)
    except Exception as e:
        logger.error(f"Необработанная ошибка: {e}", exc_info=True)
        print(f"\n❌ Критическая ошибка: {e}")
        print("Пожалуйста, проверьте файл book_tracker.log для деталей")
        sys.exit(1)


if __name__ == "__main__":
    main()
